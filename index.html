01

Scripting

Every system in Luau: server-checked gameplay, saves that don't lose data, and economies that hold up when players find the shortcuts.

RewardService.server.luau
-- every hit is checked on the server before it pays
local function onHit(player, target)
  local data = Profiles[player]
  if not data or not inReach(player, target) then return end
  local hp = target:GetAttribute("HP") - data.power
  target:SetAttribute("HP", hp)
  if hp <= 0 then
    data.coins += target:GetAttribute("Reward")
    Remotes.Reward:FireClient(player, data.coins)
  end
end
